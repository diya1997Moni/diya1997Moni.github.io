Writing: AI in marketing

How I use AI to speed up ad creative testing

Creative is the biggest lever in paid social, and also the slowest part of the workflow. AI shortens the gap between an idea and a live test.

My routine: I start from a single customer insight, ask Claude for 10 angles (benefit, objection, proof, urgency, price), then cut that to the 3 I'd actually run. Each angle gets a hook, body copy and call to action, written to the platform's character limits.

The human part matters most. AI gives volume; I choose what fits the brand, the audience and the offer. I also check every claim, because AI will happily invent a statistic.

Result: more variants tested per week, and a clearer read on which angle moved results, not just which ad.

Where AI helps performance marketers, and where it doesn't

Helps: first drafts of ad copy; turning messy exports into summaries; naming and tagging conventions; keyword grouping; drafting briefs for designers; explaining a report to a non-marketer.

Doesn't (yet): deciding budget strategy with no context; knowing your client's margins; judging brand tone without examples; verifying its own numbers.

Rule I follow: AI drafts, I decide. Anything that spends money or goes live gets human review first.

From CSV to insight: a simple AI-assisted reporting workflow

Monthly reporting eats hours. My workflow: export from the ad platform, run a small script to calculate CTR, CPC, CPA and ROAS, then use AI to draft the commentary.

The script does the maths, so the numbers are reliable. The AI explains the pattern in plain English: what moved, what likely caused it, what to test next. I edit before it reaches a client.

The scripts below are exactly what I use for the first step. They work on exported files, so there is no risk to a live account.

utm_builder.py

Consistent UTM links from a CSV, so campaign tracking stays clean.

"""Build consistent UTM-tagged URLs from a CSV of campaigns.

Input CSV columns: url, source, medium, campaign, content (optional), term (optional)
Usage: python3 utm_builder.py sample-data/campaigns.csv > tagged_urls.csv
"""
import csv
import sys
from urllib.parse import urlencode, urlparse, parse_qsl, urlunparse


def slug(value: str) -> str:
    return "_".join(value.strip().lower().split())


def tag(row: dict) -> str:
    parts = urlparse(row["url"].strip())
    query = dict(parse_qsl(parts.query))
    query.update({
        "utm_source": slug(row["source"]),
        "utm_medium": slug(row["medium"]),
        "utm_campaign": slug(row["campaign"]),
    })
    for key in ("content", "term"):
        if row.get(key, "").strip():
            query[f"utm_{key}"] = slug(row[key])
    return urlunparse(parts._replace(query=urlencode(query)))


def main(path: str) -> None:
    with open(path, newline="", encoding="utf-8") as f:
        rows = list(csv.DictReader(f))
    writer = csv.writer(sys.stdout)
    writer.writerow(["campaign", "tagged_url"])
    for row in rows:
        writer.writerow([row["campaign"], tag(row)])


if __name__ == "__main__":
    if len(sys.argv) != 2:
        sys.exit(__doc__)
    main(sys.argv[1])
ad_performance_summary.py

CTR, CPC, CPA and ROAS from an ad export, flagging campaigns to scale or review.

"""Summarise an ad platform CSV export and flag winners and underperformers.

Input CSV columns: campaign, spend, impressions, clicks, conversions, revenue
Usage: python3 ad_performance_summary.py sample-data/ads_export.csv
Works on exported files only, so it never touches a live ad account.
"""
import csv
import statistics
import sys


def ratio(a: float, b: float) -> float:
    return a / b if b else 0.0


def load(path: str) -> list[dict]:
    with open(path, newline="", encoding="utf-8") as f:
        rows = []
        for r in csv.DictReader(f):
            spend, imps = float(r["spend"]), float(r["impressions"])
            clicks, conv, rev = float(r["clicks"]), float(r["conversions"]), float(r["revenue"])
            rows.append({
                "campaign": r["campaign"],
                "spend": spend,
                "ctr": ratio(clicks, imps) * 100,
                "cpc": ratio(spend, clicks),
                "cpa": ratio(spend, conv),
                "roas": ratio(rev, spend),
            })
        return rows


def main(path: str) -> None:
    rows = load(path)
    total_spend = sum(r["spend"] for r in rows)
    median_roas = statistics.median(r["roas"] for r in rows)
    print(f"{'Campaign':<24}{'Spend':>10}{'CTR %':>8}{'CPC':>8}{'CPA':>9}{'ROAS':>7}  Flag")
    for r in sorted(rows, key=lambda r: r["roas"], reverse=True):
        flag = "scale" if r["roas"] > median_roas * 1.2 else "review" if r["roas"] < median_roas * 0.8 else ""
        print(f"{r['campaign']:<24}{r['spend']:>10.2f}{r['ctr']:>8.2f}{r['cpc']:>8.2f}{r['cpa']:>9.2f}{r['roas']:>7.2f}  {flag}")
    print(f"
Total spend: {total_spend:.2f} | Median ROAS: {median_roas:.2f}")


if __name__ == "__main__":
    if len(sys.argv) != 2:
        sys.exit(__doc__)
    main(sys.argv[1])
ab_test_calculator.py

Is your A/B test result real? Significance, lift, and the sample size you need.

"""A/B test significance and sample-size calculator (standard library only).

Usage:
  python3 ab_test_calculator.py test 120 4000 156 4000     # conv_A n_A conv_B n_B
  python3 ab_test_calculator.py size 0.03 0.20             # baseline rate, relative lift to detect
"""
import math
import sys
from statistics import NormalDist

N = NormalDist()


def z_test(conv_a, n_a, conv_b, n_b):
    p_a, p_b = conv_a / n_a, conv_b / n_b
    pooled = (conv_a + conv_b) / (n_a + n_b)
    se = math.sqrt(pooled * (1 - pooled) * (1 / n_a + 1 / n_b))
    z = (p_b - p_a) / se
    return p_a, p_b, z, 2 * (1 - N.cdf(abs(z)))


def sample_size(base, rel_lift, alpha=0.05, power=0.8):
    p1, p2 = base, base * (1 + rel_lift)
    pbar = (p1 + p2) / 2
    za, zb = N.inv_cdf(1 - alpha / 2), N.inv_cdf(power)
    top = za * math.sqrt(2 * pbar * (1 - pbar)) + zb * math.sqrt(p1 * (1 - p1) + p2 * (1 - p2))
    return math.ceil(top ** 2 / (p2 - p1) ** 2)


if __name__ == "__main__":
    a = sys.argv[1:]
    if len(a) == 5 and a[0] == "test":
        p_a, p_b, z, p = z_test(int(a[1]), int(a[2]), int(a[3]), int(a[4]))
        print(f"A: {p_a:.2%}   B: {p_b:.2%}   lift: {(p_b / p_a - 1):+.1%}")
        print(f"z = {z:.2f}   p-value = {p:.4f}")
        print("Significant at 95%." if p < 0.05 else "Not significant. Keep running or call it a tie.")
    elif len(a) == 3 and a[0] == "size":
        print(f"Needed per variant: {sample_size(float(a[1]), float(a[2])):,} visitors (95% confidence, 80% power)")
    else:
        sys.exit(__doc__)
budget_allocator.py

Models diminishing returns to suggest how to shift budget between channels, with a cap on how far it moves.

"""Reallocate a marketing budget using diminishing returns.

Assumes conversions = a * spend^e for each channel (e < 1 means each extra dollar works
a little less). 'a' is fitted from last period's spend and conversions. Each channel keeps
at least (1 - max_shift) of its current spend, and the rest is handed out in small steps to
whichever channel gains most, capped at (1 + max_shift). Treat the output as a hypothesis to
test with a small budget shift, not a guarantee.

Input CSV: channel, spend, conversions
Usage: python3 budget_allocator.py sample-data/channels.csv [elasticity=0.7] [max_shift=0.5]
"""
import csv
import sys


def allocate(rows, elasticity=0.7, max_shift=0.5, steps=400):
    total = sum(r["spend"] for r in rows)
    for r in rows:
        r["a"] = r["conversions"] / r["spend"] ** elasticity
        r["new"] = r["spend"] * (1 - max_shift)
        r["cap"] = r["spend"] * (1 + max_shift)
    step = total * max_shift / steps
    for _ in range(steps):
        open_rows = [r for r in rows if r["new"] + step <= r["cap"] + 1e-9]
        if not open_rows:
            break
        best = max(open_rows, key=lambda r: r["a"] * ((r["new"] + step) ** elasticity - r["new"] ** elasticity))
        best["new"] += step
    return rows


if __name__ == "__main__":
    if len(sys.argv) < 2:
        sys.exit(__doc__)
    e = float(sys.argv[2]) if len(sys.argv) > 2 else 0.7
    shift = float(sys.argv[3]) if len(sys.argv) > 3 else 0.5
    with open(sys.argv[1], newline="") as f:
        data = [{"channel": r["channel"], "spend": float(r["spend"]), "conversions": float(r["conversions"])}
                for r in csv.DictReader(f)]
    out = allocate(data, e, shift)
    now = sum(r["conversions"] for r in out)
    then = sum(r["a"] * r["new"] ** e for r in out)
    print(f"{'Channel':<16}{'Now':>10}{'Suggested':>12}{'Change':>10}")
    for r in out:
        print(f"{r['channel']:<16}{r['spend']:>10,.0f}{r['new']:>12,.0f}{r['new'] / r['spend'] - 1:>+10.0%}")
    print(f"
Modelled conversions: {now:,.0f} -> {then:,.0f} ({then / now - 1:+.1%}), elasticity {e}, max shift {shift:.0%}")
search_term_waste_finder.py

Finds search terms that spend but never convert and suggests negative keywords.

"""Find wasted search spend in an exported search terms report and suggest negatives.
Reads a CSV export only, so it never touches a live ad account.

Input CSV: search_term, cost, clicks, conversions
Usage: python3 search_term_waste_finder.py sample-data/search_terms.csv [min_cost=20]
"""
import csv
import sys
from collections import defaultdict

STOP = {"a", "the", "in", "for", "to", "of", "and", "near", "me"}


def main(path, min_cost):
    with open(path, newline="") as f:
        rows = [{"term": r["search_term"], "cost": float(r["cost"]), "conv": float(r["conversions"])}
                for r in csv.DictReader(f)]
    waste = [r for r in rows if r["conv"] == 0 and r["cost"] >= min_cost]
    total, wasted = sum(r["cost"] for r in rows), sum(r["cost"] for r in waste)
    print(f"Zero-conversion terms over {min_cost:.0f} spend:
")
    for r in sorted(waste, key=lambda r: -r["cost"]):
        print(f"  {r['term']:<34}{r['cost']:>8.2f}")
    print(f"
Wasted: {wasted:.2f} of {total:.2f} ({wasted / total:.0%})")

    good_words = {w for r in rows if r["conv"] > 0 for w in r["term"].split()}
    by_word = defaultdict(float)
    for r in waste:
        for w in set(r["term"].split()) - STOP - good_words:
            by_word[w] += r["cost"]
    print("
Negative keyword candidates (words that never appear in a converting term):")
    for w, c in sorted(by_word.items(), key=lambda x: -x[1])[:8]:
        print(f"  {w:<20}{c:>8.2f}")
    print("
Review before adding. A word can be harmless in one context and costly in another.")


if __name__ == "__main__":
    if len(sys.argv) < 2:
        sys.exit(__doc__)
    main(sys.argv[1], float(sys.argv[2]) if len(sys.argv) > 2 else 20)
creative_fatigue_detector.py

Spots ads whose click-through rate is decaying as frequency climbs.

"""Flag ads whose click-through rate is decaying while frequency climbs.

Input CSV: ad, date, impressions, clicks, frequency  (one row per ad per day)
Usage: python3 creative_fatigue_detector.py sample-data/creative_daily.csv
"""
import csv
import sys
from collections import defaultdict


def ctr(rows):
    imps = sum(r["imps"] for r in rows)
    return sum(r["clicks"] for r in rows) / imps if imps else 0.0


def main(path, window=4, drop_threshold=0.25):
    ads = defaultdict(list)
    with open(path, newline="") as f:
        for r in csv.DictReader(f):
            ads[r["ad"]].append({"date": r["date"], "imps": int(r["impressions"]),
                                 "clicks": int(r["clicks"]), "freq": float(r["frequency"])})
    print(f"{'Ad':<14}{'Early CTR':>10}{'Recent CTR':>12}{'Change':>9}{'Freq':>6}  Status")
    for ad, rows in ads.items():
        rows.sort(key=lambda r: r["date"])
        if len(rows) < window * 2:
            continue
        early, recent = ctr(rows[:window]), ctr(rows[-window:])
        change = recent / early - 1 if early else 0
        freq = rows[-1]["freq"]
        status = "FATIGUED: refresh" if change <= -drop_threshold and freq >= 2 else \
                 "watch" if change <= -drop_threshold / 2 else "healthy"
        print(f"{ad:<14}{early:>10.2%}{recent:>12.2%}{change:>+9.0%}{freq:>6.1f}  {status}")


if __name__ == "__main__":
    if len(sys.argv) != 2:
        sys.exit(__doc__)
    main(sys.argv[1])
ad_copy_variants.py

Ad headline and description variants from the Claude API, filtered to character limits.

"""Generate ad copy variants with the Claude API, enforcing platform character limits.

Setup:  pip install anthropic   and   export ANTHROPIC_API_KEY=your_key
Usage:  python3 ad_copy_variants.py "Padel racket store in the Philippines" "Free shipping over PHP 3000"
Always review output before it goes live.
"""
import json
import sys

import anthropic

LIMITS = {"headline": 30, "description": 90}  # Google responsive search ad limits
MODEL = "claude-sonnet-5-5"


def generate(product: str, offer: str, n: int = 5) -> dict:
    client = anthropic.Anthropic()
    prompt = (
        f"Write {n} ad headlines (max {LIMITS['headline']} characters) and {n} descriptions "
        f"(max {LIMITS['description']} characters) for: {product}. Offer: {offer}. "
        "Vary the angle: benefit, urgency, proof, price. Return JSON only: "
        '{"headlines": [...], "descriptions": [...]}'
    )
    msg = client.messages.create(model=MODEL, max_tokens=800, messages=[{"role": "user", "content": prompt}])
    return json.loads(msg.content[0].text)


def within_limits(copy: dict) -> dict:
    return {
        "headlines": [h for h in copy["headlines"] if len(h) <= LIMITS["headline"]],
        "descriptions": [d for d in copy["descriptions"] if len(d) <= LIMITS["description"]],
    }


if __name__ == "__main__":
    if len(sys.argv) != 3:
        sys.exit(__doc__)
    print(json.dumps(within_limits(generate(sys.argv[1], sys.argv[2])), indent=2))

sma_crossover_backtest.py

Backtests a moving-average strategy against buy-and-hold with no look-ahead and trading costs.

"""Backtest a moving-average crossover against buy-and-hold. Educational only.

Trades the NEXT day's return after a signal (no look-ahead) and charges costs per trade.
Past results do not predict future returns, and this is not financial advice.

Input CSV: date, close
Usage: python3 sma_crossover_backtest.py sample-data/prices.csv [fast=20] [slow=50] [cost_bps=5]
"""
import csv
import math
import statistics
import sys


def sma(xs, n, i):
    return sum(xs[i - n + 1:i + 1]) / n


def metrics(returns, label):
    equity, peak, max_dd = 1.0, 1.0, 0.0
    for r in returns:
        equity *= 1 + r
        peak = max(peak, equity)
        max_dd = min(max_dd, equity / peak - 1)
    sd = statistics.pstdev(returns)
    sharpe = statistics.mean(returns) / sd * math.sqrt(252) if sd else 0
    print(f"{label:<14} return {equity - 1:>+8.1%}   max drawdown {max_dd:>7.1%}   Sharpe {sharpe:>5.2f}")


def main(path, fast, slow, cost_bps):
    with open(path, newline="") as f:
        closes = [float(r["close"]) for r in csv.DictReader(f)]
    strat, hold, pos, trades = [], [], 0, 0
    for i in range(slow, len(closes) - 1):
        want = 1 if sma(closes, fast, i) > sma(closes, slow, i) else 0
        cost = cost_bps / 10000 if want != pos else 0
        trades += want != pos
        pos = want
        day = closes[i + 1] / closes[i] - 1
        strat.append(pos * day - cost)
        hold.append(day)
    print(f"SMA {fast}/{slow}, {cost_bps} bps per trade, {trades} trades over {len(strat)} days
")
    metrics(strat, "Crossover")
    metrics(hold, "Buy and hold")
    print("
Note: results on synthetic or past data prove nothing about the future.")


if __name__ == "__main__":
    a = sys.argv[1:]
    if not a:
        sys.exit(__doc__)
    main(a[0], int(a[1]) if len(a) > 1 else 20, int(a[2]) if len(a) > 2 else 50,
         float(a[3]) if len(a) > 3 else 5)
position_size_calculator.py

Sizes a position from account risk, entry and stop, with reward-to-risk and break-even win rate.

"""Risk-based position sizing. Educational only, not financial advice.

Usage: python3 position_size_calculator.py ACCOUNT RISK_PCT ENTRY STOP [TARGET]
Example: python3 position_size_calculator.py 10000 1 50 47 59
"""
import math
import sys


def main(account, risk_pct, entry, stop, target=None):
    risk_cash = account * risk_pct / 100
    per_share = abs(entry - stop)
    shares = math.floor(risk_cash / per_share)
    print(f"Risking {risk_pct}% = {risk_cash:,.2f}  |  stop distance {per_share:.2f}")
    print(f"Position: {shares} units, cost {shares * entry:,.2f} ({shares * entry / account:.0%} of account)")
    if target:
        rr = abs(target - entry) / per_share
        print(f"Reward:risk = {rr:.2f}:1  |  break-even win rate = {1 / (1 + rr):.0%}")


if __name__ == "__main__":
    a = sys.argv[1:]
    if len(a) not in (4, 5):
        sys.exit(__doc__)
    main(float(a[0]), float(a[1]), float(a[2]), float(a[3]), float(a[4]) if len(a) == 5 else None)
monte_carlo_portfolio.py

Simulates thousands of possible portfolio paths to show the range of outcomes and drawdown risk.

"""Monte Carlo simulation of a portfolio's possible outcomes. Educational only.

Shows the range of outcomes, not a forecast. Real returns have fatter tails than this model.

Usage: python3 monte_carlo_portfolio.py [start=10000] [annual_return=0.07] [annual_vol=0.16] [years=10] [sims=5000]
"""
import math
import random
import sys


def simulate(start, mu, vol, years, sims, seed=7):
    random.seed(seed)
    months = years * 12
    drift, sd = (mu - vol ** 2 / 2) / 12, vol / math.sqrt(12)
    finals, deep_dd = [], 0
    for _ in range(sims):
        v, peak, worst = start, start, 0.0
        for _ in range(months):
            v *= math.exp(random.gauss(drift, sd))
            peak = max(peak, v)
            worst = min(worst, v / peak - 1)
        finals.append(v)
        deep_dd += worst <= -0.30
    finals.sort()
    return finals, deep_dd / sims


if __name__ == "__main__":
    a = [float(x) for x in sys.argv[1:]] + [None] * 5
    start, mu, vol = a[0] or 10000, a[1] if a[1] is not None else 0.07, a[2] or 0.16
    years, sims = int(a[3] or 10), int(a[4] or 5000)
    finals, dd = simulate(start, mu, vol, years, sims)
    pct = lambda p: finals[int(p * (len(finals) - 1))]
    print(f"{sims:,} simulations, {years} years, {mu:.0%} return, {vol:.0%} volatility, start {start:,.0f}
")
    for p in (0.05, 0.25, 0.5, 0.75, 0.95):
        print(f"  {int(p * 100):>2}th percentile: {pct(p):>12,.0f}")
    print(f"
  Chance of ending below start: {sum(f < start for f in finals) / len(finals):.0%}")
    print(f"  Chance of a 30%+ drawdown along the way: {dd:.0%}")
